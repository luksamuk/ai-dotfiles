* Preparation

This is a lightweight Emacs configuration for work environments. It
provides essential packages, a dark theme, basic C/C++ indentation,
and AI assistance via Ellama (connecting to a local llama-swap
fleet).

To use it, copy this file to your =~/.emacs.d/init.org=, then
bootstrap it with the following =~/.emacs.d/init.el=:

#+begin_src emacs-lisp :tangle no
;; Bootstrap: tangle init.org on first load
(require 'org)
(org-babel-tangle-file
 (expand-file-name "init.org" user-emacs-directory))
(load (expand-file-name "init.el" user-emacs-directory))
#+end_src

** Bootstrapping

Tangle and compile this file on save.

#+begin_src emacs-lisp
;;;; -*- lexical-binding: t -*-

(defun tangle-init ()
  "If the current buffer is 'init.org' the code-blocks are
tangled, and the tangled file is compiled."
  (when (equal (buffer-file-name)
               (expand-file-name (concat user-emacs-directory "init.org")))
    (let ((prog-mode-hook nil))
      (org-babel-tangle))))

(add-hook 'after-save-hook 'tangle-init)
#+end_src

Some packages use =straight.el= along with =use-package=.

(defvar bootstrap-version)

(let ((bootstrap-file
       (expand-file-name "straight/repos/straight.el/bootstrap.el" user-emacs-directory))
      (bootstrap-version 5))
  (unless (file-exists-p bootstrap-file)
    (with-current-buffer
        (url-retrieve-synchronously
         "https://raw.githubusercontent.com/raxod502/straight.el/develop/install.el"
         'silent 'inhibit-cookies)
      (goto-char (point-max))
      (eval-print-last-sexp)))

* AI

** Ellama

AI assistant using local llama-swap models. The endpoint points to a
machine on the local network running the llama-swap fleet.

Models available:
- =ornith-1.0-35b= (35B MoE, agentic coding, flagship)
- =glm-4.7-flash= (30B MoE MLA, general-purpose)
- =qwen3.5-9b= / =qwen3.5-4b= (light, multimodal)
- =lfm2.5-8b-a1b= (8B MoE, general-purpose)
- =lfm2.5-2.6b= (lightweight, fast)

All models support =:think= variants for reasoning mode.

#+begin_src emacs-lisp
;; --- Configuration: adjust this IP as needed ---
(defconst my-llama-swap-endpoint "http://192.168.3.50:12434/v1/"
  "llama-swap API endpoint on the local network.")
;; ----------------------------------------------

(use-package ellama
  :init
  (setopt ellama-keymap-prefix "C-x E")
  (global-set-key (kbd "C-x e") 'ellama)
  (setopt ellama-language "Portuguese")
  (setopt ellama-auto-scroll t)
  (require 'llm-openai)
  (require 'llm-models)

  ;; Silence "not free software" warnings.
  (setopt llm-warn-on-nonfree nil)

  ;; Register local models in llm-models to avoid warnings.
  (llm-models-add
   :name "ornith-1.0-35b"
   :symbol 'ornith-1.0-35b
   :capabilities '(tool-use reasoning json-response)
   :context-length 131072
   :regex "ornith-1.0-35b")
  (llm-models-add
   :name "glm-4.7-flash"
   :symbol 'glm-4.7-flash
   :capabilities '(tool-use reasoning json-response)
   :context-length 131072
   :regex "glm-4.7-flash")
  (llm-models-add
   :name "qwen3.5-9b"
   :symbol 'qwen3.5-9b
   :capabilities '(tool-use reasoning json-response)
   :context-length 131072
   :regex "qwen3.5-9b")
  (llm-models-add
   :name "qwen3.5-4b"
   :symbol 'qwen3.5-4b
   :capabilities '(tool-use reasoning json-response)
   :context-length 131072
   :regex "qwen3.5-4b")
  (llm-models-add
   :name "lfm2.5-8b-a1b"
   :symbol 'lfm2.5-8b-a1b
   :capabilities '(tool-use reasoning json-response)
   :context-length 131072
   :regex "lfm2.5-8b-a1b")
  (llm-models-add
   :name "lfm2.5-2.6b"
   :symbol 'lfm2.5-2.6b
   :capabilities '(tool-use reasoning json-response)
   :context-length 131072
   :regex "lfm2.5-2.6b")

  (setopt ellama-naming-scheme 'ellama-generate-name-by-llm)

  ;; Main provider: Ornith (flagship, local llama-swap)
  (setopt ellama-provider
          (make-llm-openai-compatible
           :chat-model "ornith-1.0-35b:think"
           :url my-llama-swap-endpoint))

  ;; Coding provider: same as main
  (setopt ellama-coding-provider
          (make-llm-openai-compatible
           :chat-model "ornith-1.0-35b:think"
           :url my-llama-swap-endpoint))

  ;; Selectable providers (all local, via llama-swap)
  (setopt ellama-providers
	  '(("Ornith 1.0 35B" (make-llm-openai-compatible
		              :chat-model "ornith-1.0-35b:think"
		              :url my-llama-swap-endpoint))
	    ("GLM 4.7 Flash" (make-llm-openai-compatible
		              :chat-model "glm-4.7-flash:think"
		              :url my-llama-swap-endpoint))
	    ("Qwen 3.5 9B" (make-llm-openai-compatible
		            :chat-model "qwen3.5-9b:think"
		            :url my-llama-swap-endpoint))
	    ("Qwen 3.5 4B" (make-llm-openai-compatible
		            :chat-model "qwen3.5-4b:think"
		            :url my-llama-swap-endpoint))
	    ("LFM2.5 8B" (make-llm-openai-compatible
		          :chat-model "lfm2.5-8b-a1b:think"
		          :url my-llama-swap-endpoint))
	    ("LFM2.5 2.6B" (make-llm-openai-compatible
		            :chat-model "lfm2.5-2.6b:think"
		            :url my-llama-swap-endpoint))))

  (setopt ellama-chat-display-action-function #'display-buffer-full-frame)
  (setopt ellama-instant-display-action-function #'display-buffer-at-bottom)
  :config
  ;; Agentic coding: enable all tools, enforce DLP, block irreversible
  ;; actions, longer agent loops.
  (ellama-setup-agentic-coding)
  (ellama-context-header-line-global-mode +1)
  (ellama-session-header-line-global-mode +1)
  (advice-add 'pixel-scroll-precision :before #'ellama-disable-scroll)
  (advice-add 'end-of-buffer :after #'ellama-enable-scroll))
#+end_src
